=== CODE STRUCTURE TEMPLATE ===
```javascript
(function() {
    'use strict';

    // 1. CONFIGURATION
    const CONFIG = { ... };

    // 2. STATE (with explicit array initialization)
    const state = {
        objects: [],
        particles: null,
        listeners: {},
        animationId: null,
        isRunning: false,
        clock: null
    };

    // 3. ENGINE
    const engine = {
        init() {
            // Reset arrays first
            state.objects = [];

            // Setup...
            this.addListeners();
            state.isRunning = true;
            this.animate();
        },

        addListeners() {
            const onResize = () => { ... };
            const onMouseMove = (e) => { ... };

            state.listeners = { onResize, onMouseMove };

            window.addEventListener('resize', state.listeners.onResize);
            window.addEventListener('mousemove', state.listeners.onMouseMove);
        },

        animate() {
            if (!state.isRunning) return;

            const delta = Math.min(state.clock.getDelta(), 0.1);

            // Update logic...

            state.animationId = requestAnimationFrame(() => this.animate());
        },

        cleanup() {
            // 1. Cancel animation
            if (state.animationId) {
                cancelAnimationFrame(state.animationId);
                state.animationId = null;
            }

            // 2. Remove listeners
            window.removeEventListener('resize', state.listeners.onResize);
            window.removeEventListener('mousemove', state.listeners.onMouseMove);

            // 3. Dispose objects
            for (const obj of state.objects) {
                obj.geometry.dispose();
                obj.material.dispose();
                state.scene.remove(obj);
            }
            state.objects = [];

            // 4. Dispose renderer
            if (state.renderer) {
                state.renderer.dispose();
            }

            state.isRunning = false;
        },

        reset() {
            this.cleanup();
            setTimeout(() => this.init(), 100);
        }
    };

    // 4. INITIALIZATION
    document.addEventListener('DOMContentLoaded', () => {
        engine.init();
    });
})();
```
